3.11.3 Algorithm Visualizer Modules

The visualizer system implements a deterministic, discrete animation state machine. Rather than using arbitrary `setTimeout` delays during execution, algorithms execute upfront to generate a History Frame Buffer. The playback controller then navigates forward or backward through this buffer at the user's chosen animation speed.

Key References: visualizer.js and visualizer-core.js.

Source Code: Visualizer Core Animation Loop (visualizer-core.js)

// visualizer-core.js - Deterministic Step & Playback Engine
export class VisualizerCore {
  constructor(canvasContainer, options = {}) {
    this.container = typeof canvasContainer === 'string' 
      ? document.querySelector(canvasContainer) 
      : canvasContainer;
    this.speed = options.speed || 300; // Animation delay in milliseconds
    this.historyFrames = [];
    this.currentFrameIdx = 0;
    this.isPlaying = false;
    this.timer = null;
  }

  // Push a discrete state snapshot to the animation buffer
  pushFrame(snapshot) {
    this.historyFrames.push({
      array: [...snapshot.array],
      highlightIndices: [...(snapshot.highlights || [])],
      swappingIndices: [...(snapshot.swaps || [])],
      sortedIndices: [...(snapshot.sorted || [])],
      codeLine: snapshot.line || null
    });
  }

  // Continuous Playback Loop
  play() {
    if (this.isPlaying) return;
    this.isPlaying = true;
    this.runLoop();
  }

  pause() {
    this.isPlaying = false;
    clearTimeout(this.timer);
  }

  stepForward() {
    if (this.currentFrameIdx < this.historyFrames.length - 1) {
      this.currentFrameIdx++;
      this.renderFrame(this.historyFrames[this.currentFrameIdx]);
    }
  }

  stepBackward() {
    if (this.currentFrameIdx > 0) {
      this.currentFrameIdx--;
      this.renderFrame(this.historyFrames[this.currentFrameIdx]);
    }
  }

  runLoop() {
    if (!this.isPlaying) return;
    if (this.currentFrameIdx < this.historyFrames.length - 1) {
      this.stepForward();
      this.timer = setTimeout(() => this.runLoop(), this.speed);
    } else {
      this.pause();
    }
  }

  // Render DOM Bar Canvas with dynamic styles
  renderFrame(frame) {
    if (!frame || !this.container) return;
    this.container.innerHTML = '';
    const maxVal = Math.max(...frame.array, 1);

    frame.array.forEach((val, idx) => {
      const bar = document.createElement('div');
      bar.className = 'visualizer-bar';
      bar.style.height = `${(val / maxVal) * 100}%`;
      
      if (frame.sortedIndices.includes(idx)) {
        bar.classList.add('is-sorted'); // Green color
      } else if (frame.swappingIndices.includes(idx)) {
        bar.classList.add('is-swapping'); // Red color
      } else if (frame.highlightIndices.includes(idx)) {
        bar.classList.add('is-comparing'); // Yellow color
      }

      const label = document.createElement('span');
      label.innerText = val;
      bar.appendChild(label);
      this.container.appendChild(bar);
    });
  }
}

3.11.4 Practice Modules

The practice module enables learners to write and execute code directly in their browser without requiring local compiler installations. The client maps language selections to specific containerized compilers on the public Wandbox API and displays standard output, compiler warnings, or runtime errors.

Key Reference: practice-app.js.

Source Code: Wandbox Remote Compiler Client (practice-app.js)

// practice-app.js - Multi-Language Remote Compiler Relay
export class PracticeCompilerClient {
  constructor() {
    this.apiEndpoint = 'https://wandbox.org/api/compile.json';
    this.compilerByLanguage = {
      python: { compiler: 'cpython-3.13.8', file: 'main.py' },
      javascript: { compiler: 'nodejs-20.17.0', file: 'main.js' },
      java: { compiler: 'openjdk-jdk-21+35', file: 'Main.java' },
      c: { compiler: 'gcc-13.2.0-c', file: 'main.c' },
      cpp: { compiler: 'gcc-13.2.0', file: 'main.cpp' }
    };
  }

  async compileAndRun(language, sourceCode, stdinInput = '') {
    const config = this.compilerByLanguage[language];
    if (!config) throw new Error(`Unsupported programming language: ${language}`);

    const payload = {
      compiler: config.compiler,
      code: sourceCode,
      stdin: stdinInput,
      'compiler-option-raw': language === 'cpp' ? '-O2\n-std=c++20' : ''
    };

    const startTime = performance.now();
    try {
      const response = await fetch(this.apiEndpoint, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      if (!response.ok) {
        throw new Error(`Compiler API returned HTTP status ${response.status}`);
      }

      const result = await response.json();
      const elapsedMs = Math.round(performance.now() - startTime);

      return {
        status: result.status === '0' ? 'SUCCESS' : 'RUNTIME_ERROR',
        stdout: result.program_output || result.program_message || '',
        stderr: result.compiler_error || result.program_error || '',
        executionTimeMs: elapsedMs
      };
    } catch (error) {
      return {
        status: 'CONNECTION_FAILURE',
        stdout: '',
        stderr: `Failed to communicate with compiler service: ${error.message}`,
        executionTimeMs: Math.round(performance.now() - startTime)
      };
    }
  }
}

3.11.6 Authentication and Password Handling

Algorithm Hub prioritizes credential security. Passwords are never stored in plaintext; registration applies bcryptjs with 10 salt rounds. Protected endpoints verify a signed 7-day JSON Web Token (JWT). To prevent replay attacks after logout, the server computes a SHA-256 hash of the token and checks it against an active revocation table.

Key Reference: backend/server.js.

Source Code: Auth Middleware & Password Hashing (backend/server.js)

// backend/server.js - Password Hashing, JWT Issuance & Token Revocation
import jwt from 'jsonwebtoken';
import bcrypt from 'bcryptjs';
import { createHash } from 'node:crypto';

const JWT_SECRET = process.env.JWT_SECRET || 'local-development-secret-key';

// Issue 7-Day Signed Token
export const createToken = (user) => jwt.sign(
  { id: user.id, email: user.email },
  JWT_SECRET,
  { expiresIn: '7d' }
);

// Protected Route Authentication Middleware
export const requireAuth = async (req, res, next) => {
  const header = req.get('authorization') || '';
  const token = header.startsWith('Bearer ') ? header.slice(7) : '';
  if (!token) {
    return res.status(401).json({ message: 'Sign in to access your learning dashboard.' });
  }

  try {
    const payload = jwt.verify(token, JWT_SECRET);
    
    // Check if token hash exists in revoked_tokens table
    const tokenHash = createHash('sha256').update(token).digest('hex');
    const revoked = await getDb('SELECT tokenHash FROM revoked_tokens WHERE tokenHash = ?', [tokenHash]);
    if (revoked) {
      return res.status(401).json({ message: 'This session has been signed out. Please sign in again.' });
    }

    // Verify user exists in database
    const user = await getDb('SELECT id, name, email FROM users WHERE id = ?', [payload.id]);
    if (!user) {
      return res.status(401).json({ message: 'Your session is no longer valid.' });
    }

    req.user = user;
    req.token = token;
    next();
  } catch (error) {
    return res.status(401).json({ message: 'Your session is invalid or expired.' });
  }
};

// Logout Handler - Invalidate Token
app.post('/api/logout', requireAuth, async (req, res) => {
  const tokenHash = createHash('sha256').update(req.token).digest('hex');
  const payload = jwt.decode(req.token);
  const expiresAt = payload?.exp ? new Date(payload.exp * 1000).toISOString() : new Date().toISOString();

  await runDb('INSERT OR IGNORE INTO revoked_tokens (tokenHash, expiresAt) VALUES (?, ?)',
    [tokenHash, expiresAt]);

  res.json({ message: 'Signed out successfully.' });
});

3.11.7 Database Integration

The backend persists user accounts, progress, attempts, and notifications in SQLite using parameterized queries. Parameterization strictly prevents SQL injection attacks by ensuring user inputs are treated solely as literal values rather than executable SQL statements.

Key Reference: backend/server.js.

Source Code: SQLite Initialization & Parameterized Helpers (backend/server.js)

// backend/server.js - SQLite Database Driver and Table Initialization
import sqlite3 from 'sqlite3';

const DB_PATH = process.env.DB_PATH || './data/algorithm-hub.db';
const db = new sqlite3.Database(DB_PATH);

// Promise-Based Parameterized SQL Wrappers
export const runDb = (sql, params = []) => new Promise((resolve, reject) => {
  db.run(sql, params, function (err) {
    if (err) return reject(err);
    resolve({ id: this.lastID, changes: this.changes });
  });
});

export const getDb = (sql, params = []) => new Promise((resolve, reject) => {
  db.get(sql, params, (err, row) => {
    if (err) return reject(err);
    resolve(row);
  });
});

export const allDb = (sql, params = []) => new Promise((resolve, reject) => {
  db.all(sql, params, (err, rows) => {
    if (err) return reject(err);
    resolve(rows);
  });
});

// Initialize 7 SQLite Tables
export const initializeDb = async () => {
  await runDb(`
    CREATE TABLE IF NOT EXISTS users (
      id TEXT PRIMARY KEY,
      name TEXT NOT NULL,
      email TEXT NOT NULL UNIQUE,
      password TEXT NOT NULL,
      createdAt TEXT NOT NULL
    )
  `);

  await runDb(`
    CREATE TABLE IF NOT EXISTS learning_progress (
      userId TEXT NOT NULL,
      day INTEGER NOT NULL CHECK(day BETWEEN 1 AND 30),
      startedAt TEXT,
      conceptCompletedAt TEXT,
      activityCompletedAt TEXT,
      dailyQuizPassed INTEGER NOT NULL DEFAULT 0,
      latestScore REAL NOT NULL DEFAULT 0,
      completedAt TEXT,
      PRIMARY KEY (userId, day),
      FOREIGN KEY (userId) REFERENCES users(id) ON DELETE CASCADE
    )
  `);

  await runDb(`
    CREATE TABLE IF NOT EXISTS quiz_attempts (
      id TEXT PRIMARY KEY,
      userId TEXT NOT NULL,
      day INTEGER NOT NULL CHECK(day BETWEEN 1 AND 30),
      score REAL NOT NULL,
      passed INTEGER NOT NULL,
      submittedAt TEXT NOT NULL,
      FOREIGN KEY (userId) REFERENCES users(id) ON DELETE CASCADE
    )
  `);

  await runDb(`
    CREATE TABLE IF NOT EXISTS activity_attempts (
      id TEXT PRIMARY KEY,
      userId TEXT NOT NULL,
      day INTEGER NOT NULL CHECK(day BETWEEN 1 AND 30),
      answer TEXT NOT NULL,
      accepted INTEGER NOT NULL,
      submittedAt TEXT NOT NULL,
      FOREIGN KEY (userId) REFERENCES users(id) ON DELETE CASCADE
    )
  `);

  await runDb(`
    CREATE TABLE IF NOT EXISTS assessment_attempts (
      id TEXT PRIMARY KEY,
      userId TEXT NOT NULL,
      assessmentId TEXT NOT NULL,
      score REAL NOT NULL,
      passed INTEGER NOT NULL,
      submittedAt TEXT NOT NULL,
      FOREIGN KEY (userId) REFERENCES users(id) ON DELETE CASCADE
    )
  `);

  await runDb(`
    CREATE TABLE IF NOT EXISTS notifications (
      id TEXT PRIMARY KEY,
      notificationKey TEXT NOT NULL UNIQUE,
      userId TEXT NOT NULL,
      day INTEGER NOT NULL CHECK(day BETWEEN 1 AND 30),
      message TEXT NOT NULL,
      createdAt TEXT NOT NULL,
      readAt TEXT,
      FOREIGN KEY (userId) REFERENCES users(id) ON DELETE CASCADE
    )
  `);

  await runDb(`
    CREATE TABLE IF NOT EXISTS revoked_tokens (
      tokenHash TEXT PRIMARY KEY,
      expiresAt TEXT NOT NULL
    )
  `);
};
